func handleWikiPage(deps Deps) http.HandlerFunc {
    return func(w http.ResponseWriter, r *http.Request) {
        slug := strings.TrimSpace(r.URL.Query().Get("slug"))   // QUERY PARAM, not path
        if slug == "" { writeError(w, deps.Logger, 400, "slug query parameter required"); return }
        // ...
        page, err := deps.Wiki.ReadPage(slug)
        // ...
        writeJSON(w, deps.Logger, 200, WikiPage{
            Slug:        slug,
            Title:       page.Title,
            BodyMD:      page.Body,
            Frontmatter: pageFrontmatter(page),  // ← THIS is where the unversioned passthrough goes
        })
    }
}
